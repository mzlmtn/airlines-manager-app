import os import time import random import re import pandas as pd from bs4 import BeautifulSoup from selenium import webdriver from selenium.webdriver.support.ui import WebDriverWait from selenium.webdriver.support import expected_conditions as EC from selenium.webdriver.common.by import By from selenium.webdriver.chrome.options import Options def get_info_by_label(container, label_text): """ Verilen element içinde belirtilen metni (label_text) arar. O metnin bulunduğu tag'in içindeki veya bir üst kapsayıcısındaki ilk tag'inin metnini döndürür. Böylece tüm div içindeki ilk tag'ini yanlışlıkla çekmesinin önüne geçilir. """ if not container: return None # Tam metni veya bir parçasını içeren metin düğümünü Regex ile nokta atışı bul text_node = container.find(string=re.compile(label_text, re.IGNORECASE)) if text_node: parent = text_node.parent # 1. İhtimal: etiketi direkt bu elementin içindedir (Ana sayfadaki Range, Wear vs. gibi) b_tag = parent.find('b') if b_tag: return b_tag.get_text(strip=True) # 2. İhtimal: etiketi parent'ın kardeş elementindedir veya dışarıdadır (Detay sayfasındaki yapı) grandparent = parent.parent if grandparent: b_tag = grandparent.find('b') if b_tag: return b_tag.get_text(strip=True) return None def main(): # --- GITHUB SECRETS KONTROLÜ --- # Ortam değişkenlerinden kullanıcı adı ve şifreyi alıyoruz USER_EMAIL = os.environ.get("AIRLINES_USER") USER_PASS = os.environ.get("AIRLINES_PASS") if not USER_EMAIL or not USER_PASS: print("HATA: GitHub Secrets (AIRLINES_USER veya AIRLINES_PASS) bulunamadı. İşlem iptal ediliyor.") return # --- TEST / DEBUG AYARI --- # Test için kaç uçağın çekileceğini buradan ayarlayabilirsin. (Şu an ilk 5 uçağı çekecek) # Her şeyin düzgün çalıştığını gördükten sonra tüm filoyu çekmek için bunu None yap. (Örn: TEST_LIMIT = None) TEST_LIMIT = None # --- AYARLAR --- base_url = "https://tycoon.airlines-manager.com" output_excel_path = "aircrafts_output.xlsx" # Oluşturulacak Excel tablosunun sütun başlıkları columns = [ "aircraft_model", "aircraft_name", "range", "use", "cargo", "eco_seats", "business_seats", "first_seats", "hub", "wear", "age", "capacity_pax", "category", "takeoff_distance", "consumption", "speed", "purchase_price", "cumulative_result", "flight_hours" ] all_aircrafts = [] scraped_count = 0 # --- GITHUB ACTIONS İÇİN CHROME AYARLARI --- chrome_options = Options() # Sunucuda (Ubuntu runner) ekran kartı ve monitör olmadığı için bu 3 ayar ZORUNLUDUR: chrome_options.add_argument("--headless=new") chrome_options.add_argument("--no-sandbox") chrome_options.add_argument("--disable-dev-shm-usage") # Headless modda elemanların görünürlüğünü garantiye almak için ekran çözünürlüğünü sabitliyoruz: chrome_options.add_argument("--window-size=1920,1080") driver = webdriver.Chrome(options=chrome_options) # --- OTOMATİK GİRİŞ --- print("\n--- OTOMATİK GİRİŞ YAPILIYOR ---") driver.get(base_url) time.sleep(random.uniform(2, 3)) username_field = WebDriverWait(driver, 10).until( EC.presence_of_element_located((By.ID, "username")) ) # Şifreyi açık yazmak yerine Secrets'tan gelen değişkeni veriyoruz human_type(username_field, USER_EMAIL) time.sleep(random.uniform(0.5, 1.5)) password_field = driver.find_element(By.ID, "password") # Şifreyi açık yazmak yerine Secrets'tan gelen değişkeni veriyoruz human_type(password_field, USER_PASS) time.sleep(random.uniform(0.5, 1.0)) login_btn = driver.find_element(By.ID, "loginSubmit") login_btn.click() WebDriverWait(driver, 15).until( EC.url_contains("/home") ) print("Giriş başarılı!") time.sleep(random.uniform(2, 3)) print("Veri kazıma işlemi başlıyor...\n") page_number = 1 stop_scraping = False while not stop_scraping: list_url = f"{base_url}/aircraft?page={page_number}" print(f"--- SAYFA {page_number} taranıyor: {list_url} ---") driver.get(list_url) time.sleep(random.uniform(1.7, 2.3)) soup = BeautifulSoup(driver.page_source, 'html.parser') aircraft_boxes = soup.find_all('div', class_='aircraftListBox') if not aircraft_boxes: print(f"Sayfa {page_number}'de uçak bulunamadı. Tarama tamamlandı!") break print(f"Sayfa {page_number} üzerinde {len(aircraft_boxes)} adet uçak bulundu. Bilgiler çekiliyor...") # O sayfadaki uçakların detaylarına gitmeden önce ana bilgilerini tuttuğumuz geçici liste page_aircrafts_data = [] for box in aircraft_boxes: # Test limiti kontrolü if TEST_LIMIT is not None and scraped_count >= TEST_LIMIT: stop_scraping = True break aircraft_data = {col: None for col in columns} # 1. aircraft_model (Sondaki / işaretini ve boşlukları temizleyerek) title_div = box.find('div', class_='title') if title_div: main_span = title_div.find('span', recursive=False) if main_span and main_span.contents: aircraft_data["aircraft_model"] = str(main_span.contents[0]).replace('/', '').strip() # 2. aircraft_name name_span = box.find('span', class_='editAircraftName') if name_span and name_span.find('span'): aircraft_data["aircraft_name"] = name_span.find('span').get_text(strip=True) # Detay sayfası için URL'yi arka planda geçici tutuyoruz btn_detail = box.find('a', class_='BtnDetailAvion') detail_url = base_url + btn_detail['href'] if btn_detail else None aircraft_data["_detail_url"] = detail_url content_div = box.find('div', class_='content') # 3. range aircraft_data["range"] = get_info_by_label(content_div, "Range") # 4. use aircraft_data["use"] = get_info_by_label(content_div, "Use") # 5. cargo aircraft_data["cargo"] = get_info_by_label(content_div, "Cargo") # 6, 7, 8. eco_seats, business_seats, first_seats seats_raw = get_info_by_label(content_div, "Seats") if seats_raw: # Örnek Text Formatı: 492 (447/36/9) match = re.search(r"\((\d+)/(\d+)/(\d+)\)", seats_raw) if match: aircraft_data["eco_seats"] = match.group(1) aircraft_data["business_seats"] = match.group(2) aircraft_data["first_seats"] = match.group(3) # 9. hub hub_raw = get_info_by_label(content_div, "Hub") if hub_raw: aircraft_data["hub"] = hub_raw.replace('/', '').strip() # 10. wear aircraft_data["wear"] = get_info_by_label(content_div, "Wear") # 11. age aircraft_data["age"] = get_info_by_label(content_div, "Age") page_aircrafts_data.append(aircraft_data) scraped_count += 1 # --- DETAY SAYFALARINI ZİYARET ETME --- for idx, data in enumerate(page_aircrafts_data): detail_link = data.pop("_detail_url", None) # Geçici url'yi sözlükten çıkar ve kullan if not detail_link: all_aircrafts.append(data) continue print(f" -> Uçak {idx + 1}/{len(page_aircrafts_data)} detayı çekiliyor: {data['aircraft_name']}") driver.get(detail_link) time.sleep(random.uniform(1.0, 1.3)) # Rastgele güvenlik gecikmesi detail_soup = BeautifulSoup(driver.page_source, 'html.parser') # 12. capacity_pax data["capacity_pax"] = get_info_by_label(detail_soup, "Capacity") # 13. category right_panel = detail_soup.find('div', id='aircraftCharacteristicRight') if right_panel: cat_img = right_panel.find('img', src=re.compile(r'/images/icons20/cat\d+\.png')) if cat_img: match = re.search(r'cat(\d+)\.png', cat_img['src']) if match: data["category"] = match.group(1) # 14. takeoff_distance data["takeoff_distance"] = get_info_by_label(detail_soup, "Takeoff distance") # 15. consumption data["consumption"] = get_info_by_label(detail_soup, "Consumption") # 16. speed data["speed"] = get_info_by_label(detail_soup, "Speed :") # 17. purchase_price data["purchase_price"] = get_info_by_label(detail_soup, "Purchase price") # 18. cumulative_result data["cumulative_result"] = get_info_by_label(detail_soup, "Cumulative result") # 19. flight_hours data["flight_hours"] = get_info_by_label(detail_soup, "Flight hours") # Tüm verileri toplanmış uçağı ana listeye ekle all_aircrafts.append(data) if not stop_scraping: page_number += 1 # Tüm döngü bitti, Tarayıcıyı kapat. driver.quit() print("\nVeri çekme işlemi tamamen bitti. Excel dosyası oluşturuluyor...") # Verileri DataFrame'e çevirip Excel'e kaydet df = pd.DataFrame(all_aircrafts, columns=columns) try: df.to_excel(output_excel_path, index=False) print(f"BAŞARILI! Toplam {len(df)} uçak '{output_excel_path}' dosyasına kaydedildi.") except Exception as e: print(f"Dosya kaydedilirken hata oluştu: {e}") def human_type(element, text): """Her karakteri rastgele gecikmeyle yazar.""" for char in text: element.send_keys(char) time.sleep(random.uniform(0.05, 0.15)) if __name__ == "__main__": main()
